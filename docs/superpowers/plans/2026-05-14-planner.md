# Planner Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the hand-rolled feature→stage gating in `gui/run_dispatch.py` with a thin planner module (`gui/planner/`) that walks a declared feature/stage dependency graph. v1 emits the same `argparse.Namespace` `main.py` consumes today, so the migration is bounded to the GUI dispatcher.

**Architecture:** New `gui/planner/` package with four files (`stages.py`, `registry.py`, `plan.py`, `emit.py`) sitting between the GUI and `main.py`. `STAGES` + `CLUSTERS` are data; `plan_from_prefs` is a pure function; `emit.to_namespace` is the migration adapter. Old hand-rolled functions get deleted once the planner produces an equivalent `Namespace`.

**Tech Stack:** Python 3.10, dataclasses (frozen), pure-stdlib for the planner logic, pytest for tests. No new external dependencies.

**Spec:** [`docs/superpowers/specs/2026-05-14-planner-design.md`](../specs/2026-05-14-planner-design.md). **Upstream reference:** [`docs/feature_dependencies.md`](../../feature_dependencies.md).

```python
from gui.planner.stages import Stage, FeatureCluster, Plan


def test_stage_is_frozen_dataclass():
    s = Stage(name="x", consumes=("a",), produces=("b",), run=lambda: None)
    with pytest.raises(FrozenInstanceError):
        s.name = "y"


def test_stage_fields():

    p = Plan(
        stages_to_run=("a", "b"),
        output_csvs=frozenset({"features"}),
        column_drops={"features": frozenset({"X"})},
        stage_configs={"a": {"k": "v"}},
    )
    with pytest.raises(FrozenInstanceError):
        p.stages_to_run = ("c",)

    """column_drops and stage_configs must be coerced to immutable types
    so two Plan instances can't accidentally share state."""
    p = Plan(stages_to_run=(), output_csvs=frozenset(),
             column_drops={}, stage_configs={})

    with pytest.raises(FrozenInstanceError):
```

```python
"""Frozen dataclasses for the planner. No logic — just types.

See docs/superpowers/specs/2026-05-14-planner-design.md §5.
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Callable


@dataclass(frozen=True)
class Stage:
    """One unit of pipeline execution.

    consumes/produces are artifact-name tuples used by the planner's
    DAG closure (plan.py) to determine which stages must run to satisfy
    an active cluster's `requires_stages`.
    """
    name: str
    consumes: tuple[str, ...]
    produces: tuple[str, ...]
    run: Callable


@dataclass(frozen=True)
class FeatureCluster:
    """A group of user-facing columns sharing a stage and a toggle group.

    A cluster is active iff:
    - `output_csv` is among the user's requested outputs, AND
    - any key in `toggle_keys` is True in prefs["features"].
    """
    name: str
    toggle_keys: tuple[str, ...]
    output_csv: str   # "features" | "syllables" | "phonemes" | "documents_summary"
    requires_stages: tuple[str, ...]
    emits_columns: tuple[str, ...]


@dataclass(frozen=True)
class Plan:
    """The output of plan_from_prefs — what to run, what to write."""
    stages_to_run: tuple[str, ...]
    output_csvs: frozenset[str]
    column_drops: dict[str, frozenset[str]]
    stage_configs: dict[str, dict[str, Any]]
```

```python
"""Sanity / drift tests over STAGES. CLUSTERS-specific tests in test_cluster_registry
(Task 3); full algorithm tests in test_plan.py (Task 4)."""
from gui.planner.registry import STAGES


def test_stage_names_are_unique():
    names = [s.name for s in STAGES]
    assert len(names) == len(set(names)), \
        f"Duplicate stage names: {[n for n in names if names.count(n) > 1]}"


def test_every_consumed_artifact_is_produced_by_some_stage():
    """The DAG must close: every artifact consumed by some stage must
    be produced by some other stage OR be a root input ('text', 'audio',
    'prepared_text' — the latter is produced by transcript_prep, which
    we treat as implicit since it always runs before MFA)."""
    ROOTS = {"text", "audio", "prepared_text"}
    produced = set()
    for s in STAGES:
        produced.update(s.produces)
    consumed = set()
    for s in STAGES:
        consumed.update(s.consumes)
    orphan = consumed - produced - ROOTS
    # Strip namespaced sub-artifacts like word_df.Zipf_Freq — those are produced
    # by a stage modifying the word_df. The base name "word_df" is what matters
    # for consumes-resolution.
    orphan = {a for a in orphan if "." not in a}
    assert not orphan, f"Stage consumes unproduced artifact(s): {orphan}"


def test_each_stage_produces_at_least_one_artifact():
    for s in STAGES:
        assert s.produces, f"Stage {s.name!r} produces nothing"


def test_shared_stages_present():
    names = {s.name for s in STAGES}
    assert "spacy_parse" in names
    assert "mfa_align" in names
    assert "praat_contours" in names


def test_subword_stages_present():
    names = {s.name for s in STAGES}
    assert "syllabify_pairs" in names
    assert "syllable_prosody" in names


def test_doc_stages_present():
    names = {s.name for s in STAGES}
    for n in ("doc_counts", "doc_aggregates", "doc_lex_diversity",
              "doc_readability", "doc_liwc", "doc_category_proportions",
              "doc_prosody", "doc_articulation_rate"):
        assert n in names, f"missing doc stage {n}"
```

```python
"""plan_from_prefs(prefs) -> Plan.

Pure function: deterministic, no I/O.

Algorithm (see spec §6):
  1. Determine active output CSVs from prefs["output"].
  2. For each cluster, mark active iff its output_csv is wanted AND any
     of its toggle_keys is True in prefs["features"] (or prefs["output"]
     for output-keyed clusters like doc_counts, doc_lex_diversity, etc).
  3. Union the requires_stages of active clusters.
  4. Close the set under stage.consumes ↔ stage.produces.
  5. Topologically sort.
  6. Compute per-CSV column-drop sets.
  7. Build stage_configs dict.
"""
from __future__ import annotations

from typing import Any

from gui.planner.registry import STAGES, CLUSTERS
from gui.planner.stages import Plan, Stage, FeatureCluster


# Output toggles live in prefs["output"]. Toggle-key prefixes used to figure
# out which prefs section to read for cluster activation.
_OUTPUT_KEYED_TOGGLES = frozenset({
    # doc-level output-keyed cluster toggle_keys (these live in prefs["output"], not prefs["features"])
    "out_count_tokens", "out_count_content_tokens",
    "out_count_sentences", "out_count_paragraphs",
    "doc_aggregates", "doc_lex_diversity", "doc_readability",
    "doc_category_proportions",
})

_OUTPUT_CSV_MAP = {
    "word_level_csv":   "features",
    "syllables_csv":    "syllables",
    "phonemes_csv":     "phonemes",
    "document_summary": "documents_summary",
}


def _active_output_csvs(prefs: dict) -> frozenset[str]:
    out = prefs.get("output", {})
    return frozenset(
        csv for prefs_key, csv in _OUTPUT_CSV_MAP.items()
        if out.get(prefs_key, False)
    )


def _toggle_value(prefs: dict, key: str) -> bool:
    """Read a toggle from the right prefs section."""
    if key in _OUTPUT_KEYED_TOGGLES:
        return bool(prefs.get("output", {}).get(key, False))
    return bool(prefs.get("features", {}).get(key, False))


def _active_clusters(prefs: dict, active_csvs: frozenset[str]) -> list[FeatureCluster]:
    actives = []
    for c in CLUSTERS:
        if c.output_csv not in active_csvs:
            continue
        if any(_toggle_value(prefs, k) for k in c.toggle_keys):
            actives.append(c)
    return actives


def _stage_by_name() -> dict[str, Stage]:
    return {s.name: s for s in STAGES}


def _close_under_consumes(required: set[str]) -> set[str]:
    """Walk STAGES backward: for each artifact in required stages' consumes,
    pull in any Stage that produces that artifact. Repeat until stable.

    Artifacts with a dot (e.g. 'word_df.Zipf_Freq') are namespaced — strip
    the suffix; the base artifact is what's resolved.
    """
    by_name = _stage_by_name()
    artifacts_produced_by: dict[str, list[str]] = {}
    for s in STAGES:
        for a in s.produces:
            base = a.split(".", 1)[0]
            artifacts_produced_by.setdefault(base, []).append(s.name)

    closed = set(required)
    while True:
        added = set()
        for name in list(closed):
            s = by_name[name]
            for art in s.consumes:
                base = art.split(".", 1)[0]
                for producer in artifacts_produced_by.get(base, ()):
                    if producer not in closed:
                        added.add(producer)
        if not added:
            break
        closed |= added
    return closed


def _topo_sort(stage_names: set[str]) -> tuple[str, ...]:
    """Topological sort by consumes/produces. Tiebreak by registry order
    for stable output."""
    by_name = _stage_by_name()
    registry_order = {s.name: i for i, s in enumerate(STAGES)}

    produced_by: dict[str, str] = {}
    for s in STAGES:
        for a in s.produces:
            base = a.split(".", 1)[0]
            # First producer of the base artifact wins for the topo edge.
            # (When multiple stages produce sub-namespaces of word_df, this
            # is fine: spacy_parse produces word_df, others mutate it.)
            produced_by.setdefault(base, s.name)

    in_set = set(stage_names)
    deps: dict[str, set[str]] = {n: set() for n in in_set}
    for n in in_set:
        for art in by_name[n].consumes:
            base = art.split(".", 1)[0]
            producer = produced_by.get(base)
            if producer and producer in in_set and producer != n:
                deps[n].add(producer)

    ordered: list[str] = []
    remaining = set(in_set)
    while remaining:
        ready = sorted(
            [n for n in remaining if not (deps[n] - set(ordered))],
            key=lambda n: registry_order[n],
        )
        if not ready:
            raise ValueError(f"Cycle in stage dependencies among {remaining}")
        ordered.extend(ready)
        remaining -= set(ready)
    return tuple(ordered)


def _column_drops(active_clusters: list[FeatureCluster],
                  active_csvs: frozenset[str]) -> dict[str, frozenset[str]]:
    """For each active CSV, columns from inactive clusters with that CSV
    go in the drop set. Active-cluster columns survive."""
    active_cluster_set = set(id(c) for c in active_clusters)
    drops: dict[str, set[str]] = {csv: set() for csv in active_csvs}
    for c in CLUSTERS:
        if c.output_csv not in active_csvs:
            continue
        if id(c) in active_cluster_set:
            continue
        drops[c.output_csv].update(c.emits_columns)
    return {csv: frozenset(cols) for csv, cols in drops.items()}


def _stage_configs(prefs: dict, stages_to_run: tuple[str, ...]) -> dict[str, dict[str, Any]]:
    """Per-stage parameter dict; only includes stages that need config."""
    configs: dict[str, dict[str, Any]] = {}
    if "praat_contours" in stages_to_run or "word_prosody" in stages_to_run \
            or "syllable_prosody" in stages_to_run or "doc_prosody" in stages_to_run:
        pros = prefs.get("prosody", {})
        configs["praat_contours"] = {
            "pitch_floor":     float(pros.get("pitch_floor_hz", 75.0)),
            "pitch_ceiling":   float(pros.get("pitch_ceiling_hz", 500.0)),
        }
        configs["word_prosody"] = {
            "pitch_units":     pros.get("pitch_units", "semitones"),
        }
        configs["syllable_prosody"] = {
            "pitch_units":     pros.get("pitch_units", "semitones"),
        }
        configs["doc_prosody"] = {
            "pitch_units":     pros.get("pitch_units", "semitones"),
            "pause_threshold": float(pros.get("pause_threshold_s", 0.25)),
        }
    if "llm_predict" in stages_to_run:
        configs["llm_predict"] = {
            "llm_model": prefs.get("predictability", {}).get("llm_model", "") or None,
        }
    return configs


def plan_from_prefs(prefs: dict) -> Plan:
    """Build the Plan from a user-prefs dict. Pure function; deterministic."""
    active_csvs = _active_output_csvs(prefs)
    active = _active_clusters(prefs, active_csvs)
    required: set[str] = set()
    for c in active:
        required.update(c.requires_stages)
    closed = _close_under_consumes(required)
    ordered = _topo_sort(closed)
    drops = _column_drops(active, active_csvs)
    configs = _stage_configs(prefs, ordered)
    return Plan(stages_to_run=ordered, output_csvs=active_csvs,
                column_drops=drops, stage_configs=configs)
```

```python
"""Adapter: Plan → argparse.Namespace that main.py consumes.

Single source of truth for the no_* flag mappings. Refactor in v2 when
main.py learns to consume Plans directly.

See spec §7.
"""
from __future__ import annotations

import argparse

from gui.planner.stages import Plan


_WORD_STAGE_FLAGS = {
    # stage name -> attr name on the namespace (no_X is True iff stage NOT in plan)
    "spacy_parse":    "no_parse",
    "mfa_align":      "no_align",
    "praat_contours": None,    # implicit; no flag in main.py
    "word_freq":      "no_freq",
    "ewfg_freq":      "no_edu_freq",
    "gpt2_surprisal": "no_surprisal",
    "llm_predict":    "no_llm_cloze",
    "sem_domain":     "no_sem_domain",
    "word_prosody":   "no_prosody",
}


def _enabled_ewfg_bands(features: dict) -> str | None:
    bands = [
        suf for suf, key in [
            ("k2", "freq_ewfg_k2"), ("3to5", "freq_ewfg_3to5"),
            ("6to8", "freq_ewfg_6to8"), ("9to12", "freq_ewfg_9to12"),
            ("overall", "freq_ewfg_overall"),
        ] if features.get(key, False)
    ]
    return ",".join(bands) if bands else None


def _categories_for_level(prefs: dict, level: str) -> str | None:
    backend = prefs.get("semantic", {}).get("backend", "auto")
    if backend == "auto":
        # Defer to runtime detection; emit None so main.run() falls back.
        return None
    key = f"{'doc' if level == 'document' else 'word'}_categories_{backend}"
    cats = prefs.get("semantic", {}).get(key, [])
    return ",".join(cats) if cats else None


def to_namespace(plan: Plan, prefs: dict, level: str) -> argparse.Namespace:
    """Build the argparse.Namespace for main.run_word_level_batch (level='word')
    or main.run_doc_level_batch (level='doc')."""
    features = prefs.get("features", {})
    output = prefs.get("output", {})
    shared = prefs.get("shared", {})

    ns = argparse.Namespace()

    # I/O folders (passthrough from prefs)
    ns.transcripts_folder = shared.get("last_transcripts_folder", "")
    ns.audio_folder       = shared.get("last_audio_folder", "")
    ns.output_folder      = shared.get("last_output_folder", "")
    ns.analyzer           = shared.get("analyzer_name", "")
    ns.text_extensions    = "txt,tsv,md"
    ns.audio_extensions   = "wav,flac,mp3"
    ns.save_prefs         = False

    # Word-stage no_* flags
    in_plan = set(plan.stages_to_run)
    for stage_name, attr in _WORD_STAGE_FLAGS.items():
        if attr is None:
            continue
        setattr(ns, attr, stage_name not in in_plan)

    # Sidecar / output flags
    ns.no_word_csv  = "features" not in plan.output_csvs
    ns.no_syllables = "syllables" not in plan.output_csvs
    ns.no_phonemes  = "phonemes" not in plan.output_csvs

    # Pass-throughs that main.py reads directly from args
    ns.feature_toggles = dict(features)
    ns.output_toggles  = dict(output)
    ns.edu_bands       = _enabled_ewfg_bands(features)
    ns.sem_dict        = prefs.get("semantic", {}).get("backend", "auto")
    ns.sem_categories  = _categories_for_level(prefs, level)
    ns.llm_model       = prefs.get("predictability", {}).get("llm_model", "") or None
    ns.edu_freq_lookup = None

    # Prosody runtime params
    pros = prefs.get("prosody", {})
    ns.pitch_units     = pros.get("pitch_units", "semitones")
    ns.pitch_floor     = float(pros.get("pitch_floor_hz", 75.0))
    ns.pitch_ceiling   = float(pros.get("pitch_ceiling_hz", 500.0))
    ns.pause_threshold = float(pros.get("pause_threshold_s", 0.25))

    # Doc-level extra flags
    if level == "doc":
        ns.no_doc_counts          = "doc_counts" not in in_plan
        ns.no_doc_mean_zipf       = ("doc_aggregates" not in in_plan
                                      or not features.get("freq_subtlex", False))
        ns.no_doc_mean_edu_zipf   = ("doc_aggregates" not in in_plan
                                      or _enabled_ewfg_bands(features) is None)
        ns.no_doc_mean_surprisal  = ("doc_aggregates" not in in_plan
                                      or not features.get("pred_surprisal_gpt2", False))
        ns.no_doc_mean_llm_cloze  = ("doc_aggregates" not in in_plan or not (
            features.get("pred_surprisal_llm", False)
            or features.get("pred_cloze", False)
            or features.get("pred_topmatch", False)))
        ns.no_doc_ttr             = "doc_lex_diversity" not in in_plan
        ns.no_doc_mtld            = "doc_lex_diversity" not in in_plan
        ns.no_doc_flesch_ease     = "doc_readability"   not in in_plan
        ns.no_doc_fk_grade        = "doc_readability"   not in in_plan
        ns.no_doc_liwc_authentic  = "doc_liwc"          not in in_plan
        ns.no_doc_liwc_analytic   = "doc_liwc"          not in in_plan
        ns.no_doc_liwc_tone       = "doc_liwc"          not in in_plan
        ns.no_doc_liwc_clout      = "doc_liwc"          not in in_plan
        ns.no_doc_category_proportions = "doc_category_proportions" not in in_plan
        ns.no_doc_prosody         = "doc_prosody"       not in in_plan

    return ns
```

```python
def _prepare_runtime_args(ns: argparse.Namespace) -> argparse.Namespace:
    """Copy + convert edu_bands list to comma-joined string (the planner
    already emits a string, but this stays for CLI-path compatibility)."""
    rt = copy.copy(ns)

def run(progress_cb: Callable[[int, int, str], None],
        done_cb: Callable[[str], None],
        session_api_key: str,
        in_thread: bool = True) -> None:
    """Dispatch a run in a background thread (or inline when in_thread=False)."""

    def _go():
        import os
        if session_api_key:
            os.environ["OPENAI_API_KEY"] = session_api_key
        prefs = load_preferences()
        plan = planner.plan_from_prefs(prefs)
        run_ts = _dt.datetime.now().strftime('%Y-%m-%d_%H-%M-%S')
        try:
            word_outputs = {"features", "syllables", "phonemes"} & plan.output_csvs
            if word_outputs:
                from main import run_word_level_batch
                ns = _prepare_runtime_args(planner.to_namespace(plan, prefs, level="word"))
                ns.run_ts = run_ts
                ns.skip_no_match = False
                run_word_level_batch(ns, on_pair_done=progress_cb)
            if "documents_summary" in plan.output_csvs:
                from main import run_doc_level_batch
                ns = _prepare_runtime_args(planner.to_namespace(plan, prefs, level="doc"))
                ns.run_ts = run_ts
                ns.skip_no_match = bool(word_outputs)
                run_doc_level_batch(ns, on_pair_done=progress_cb)
            done_cb("done")
        except Exception as exc:
            done_cb(f"error: {exc}")
        finally:
            if session_api_key:
                os.environ.pop("OPENAI_API_KEY", None)

    if in_thread:
        threading.Thread(target=_go, daemon=True).start()
    else:
        _go()
```

```python
"""Schema-drift guard. Every prefs key that controls feature behavior
must be referenced by some FeatureCluster's toggle_keys — otherwise
the planner silently ignores it."""
from gui.planner.registry import CLUSTERS
from utils.preferences import _FEATURES_DEFAULT_V6, _OUTPUT_DEFAULT_V6


# Keys that are intentionally NOT in any cluster (config knobs, not
# feature toggles).
_INTENTIONALLY_UNROUTED_FEATURES: set[str] = set()  # currently empty

# Output keys NOT routed via clusters (the master output toggles that
# select which CSV gets written are handled by _OUTPUT_CSV_MAP in
# plan.py, not by clusters).
_OUTPUT_KEYS_VIA_CSV_MAP: set[str] = {
    "word_level_csv", "syllables_csv", "phonemes_csv", "document_summary",
}


def _all_cluster_toggle_keys() -> set[str]:
    return {k for c in CLUSTERS for k in c.toggle_keys}


def test_every_feature_pref_key_is_in_some_cluster():
    routed = _all_cluster_toggle_keys()
    feature_keys = set(_FEATURES_DEFAULT_V6.keys())
    unrouted = feature_keys - routed - _INTENTIONALLY_UNROUTED_FEATURES
    assert not unrouted, (
        f"Feature pref keys not referenced by any cluster (the planner will "
        f"silently ignore them when deciding what runs): {sorted(unrouted)}"
    )


def test_every_output_pref_key_is_either_master_csv_or_in_some_cluster():
    routed = _all_cluster_toggle_keys()
    output_keys = set(_OUTPUT_DEFAULT_V6.keys())
    unrouted = output_keys - routed - _OUTPUT_KEYS_VIA_CSV_MAP
    assert not unrouted, (
        f"Output pref keys not referenced by any cluster or master CSV map "
        f"(planner ignores them): {sorted(unrouted)}"
    )


def test_no_cluster_references_nonexistent_pref_key():
    """The reverse direction: cluster toggle_keys must exist in the prefs schema."""
    all_pref_keys = set(_FEATURES_DEFAULT_V6.keys()) | set(_OUTPUT_DEFAULT_V6.keys())
    routed = _all_cluster_toggle_keys()
    phantom = routed - all_pref_keys
    assert not phantom, (
        f"Clusters reference pref keys not in the v6 schema "
        f"(typo or stale registry): {sorted(phantom)}"
    )
```
