# Batch 3 Implementation Plan — Word/Doc Separation, Modal Categories, Per-Column Toggles

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to execute task-by-task.

**Goal:** Split the Features page into Word-level and Document-level sidebar pages, replace the inline category picker with two modal dialogs (one per level, independent selections), expand 4 single-toggle feature blocks (Structural, Acoustic, Composite psych, Doc Counts) into per-column checklists, wrap each subsection in a card container, and migrate preferences from v3 → v4.

**Architecture:** `gui/section_features.py` splits into `gui/section_word_features.py` + `gui/section_doc_features.py`. A new `gui/category_dialog.py` wraps `CategoryPicker` in a `QDialog`. `CategoryPicker` gains a `level` parameter and reads/writes level-specific prefs keys. CSV writers in `main.py` drop columns whose per-column toggles are off. Sidebar grows to 5 items.

**Tech Stack:** Same as batch 2 — PySide6, pytest-qt, tiktoken (already installed). No new dependencies.

**Spec:** `docs/superpowers/specs/2026-05-11-batch-3-design.md`

```python
SCHEMA_VERSION = SCHEMA_VERSION_V4
DEFAULT_PREFS: dict[str, Any] = DEFAULT_PREFS_V4


def _migrate_v3_into_v4(merged: dict, v3: dict) -> None:
    """Convert v3-shape `v3` into v4-shape `merged` (mutates merged)."""
    # shared — same keys
    for k, v in v3.get("shared", {}).items():
        merged["shared"][k] = v

    # predictability — same keys
    for k, v in v3.get("predictability", {}).items():
        merged["predictability"][k] = v

    # semantic — backend stays; categories duplicated to both levels
    v3_sem = v3.get("semantic", {})
    merged["semantic"]["backend"] = v3_sem.get("backend", "auto")
    cats_liwc = v3_sem.get("categories_liwc", list(_SOP_CATEGORIES_LIWC))
    cats_empath = v3_sem.get("categories_empath", list(_SOP_CATEGORIES_EMPATH))
    merged["semantic"]["word_categories_liwc"] = list(cats_liwc)
    merged["semantic"]["doc_categories_liwc"] = list(cats_liwc)
    merged["semantic"]["word_categories_empath"] = list(cats_empath)
    merged["semantic"]["doc_categories_empath"] = list(cats_empath)

    # features — most keys unchanged; expand 3 collapsed parents into per-column
    v3_feat = v3.get("features", {})
    f = merged["features"]
    for k in (
        "lexling_tokens", "lexling_pos", "lexling_morph", "lexling_dep",
        "freq_subtlex",
        "freq_ewfg_k2", "freq_ewfg_3to5", "freq_ewfg_6to8",
        "freq_ewfg_9to12", "freq_ewfg_overall",
        "pred_surprisal_gpt2", "pred_surprisal_llm",
        "pred_cloze", "pred_topmatch",
        "sem_indicators",
    ):
        if k in v3_feat:
            f[k] = v3_feat[k]
    struct_on = bool(v3_feat.get("structural", True))
    for k in ("feat_struct_doc_id", "feat_struct_paragraph_id",
              "feat_struct_sentence_id", "feat_struct_sentence_text",
              "feat_struct_char_start", "feat_struct_char_end",
              "feat_struct_term_id", "feat_struct_token_id"):
        f[k] = struct_on
    acoustic_on = bool(v3_feat.get("acoustic_boundaries", True))
    for k in ("feat_acoustic_onset", "feat_acoustic_end",
              "feat_acoustic_word_duration", "feat_acoustic_word_gap"):
        f[k] = acoustic_on
    psych_on = bool(v3_feat.get("sem_composite_psych", True))
    for k in ("feat_psych_authentic", "feat_psych_analytic",
              "feat_psych_tone", "feat_psych_clout"):
        f[k] = psych_on

    # output — most keys unchanged; expand doc_counts to per-column
    v3_out = v3.get("output", {})
    o = merged["output"]
    for k in ("word_level_csv", "document_summary",
              "doc_aggregates", "doc_lex_diversity",
              "doc_readability", "doc_category_proportions"):
        if k in v3_out:
            o[k] = v3_out[k]
    counts_on = bool(v3_out.get("doc_counts", True))
    for k in ("out_count_tokens", "out_count_content_tokens",
              "out_count_sentences", "out_count_paragraphs"):
        o[k] = counts_on


def _deep_merge_v4(into: dict, src: dict) -> None:
    """Merge a v4-shape src dict into into. Inner dicts merge by key."""
    for key, value in src.items():
        if key in ("shared", "features", "predictability", "semantic", "output") \
                and isinstance(value, dict):
            existing = into.get(key, {})
            if not isinstance(existing, dict):
                existing = {}
            for sub_key, sub_value in value.items():
                existing[sub_key] = sub_value
            into[key] = existing
        else:
            into[key] = value
```

Then update `load_preferences`:

```python
def load_preferences() -> dict[str, Any]:
    path = get_preferences_path()
    merged: dict[str, Any] = deepcopy(DEFAULT_PREFS)

    if not path.exists():
        return merged
    try:
        with path.open("r", encoding="utf-8") as f:
            user = json.load(f)
    except (json.JSONDecodeError, OSError):
        return merged
    if not isinstance(user, dict):
        return merged

    version = user.get("schema_version", 1)
    if version == SCHEMA_VERSION_V4:
        _deep_merge_v4(merged, user)
    elif version == 3:
        _migrate_v3_into_v4(merged, user)
        _write(path, merged)
    elif version == 2:
        # chain v2 → v3 → v4
        v3_intermediate = deepcopy(DEFAULT_PREFS_V3)
        _migrate_v2_into_v3(v3_intermediate, user)
        _migrate_v3_into_v4(merged, v3_intermediate)
        _write(path, merged)
    else:
        # v1 → v2 → v3 → v4
        v2_intermediate = deepcopy(_DEFAULT_PREFS_V2)
        _migrate_v1_into(v2_intermediate, user)
        v3_intermediate = deepcopy(DEFAULT_PREFS_V3)
        _migrate_v2_into_v3(v3_intermediate, v2_intermediate)
        _migrate_v3_into_v4(merged, v3_intermediate)
        _write(path, merged)

    return merged
```

```python
def _word_columns_to_drop(features: dict[str, bool]) -> set[str]:

    struct_map = {
        "feat_struct_doc_id": "doc_id",
        "feat_struct_paragraph_id": "paragraph_id",
        "feat_struct_sentence_id": "sentence_id",
        "feat_struct_sentence_text": "sentence",

        "feat_struct_char_end": "end",
        "feat_struct_term_id": "term_id",
        "feat_struct_token_id": "token_id",
    }
    acoustic_map = {
        "feat_acoustic_onset": "Onset",
        "feat_acoustic_end": "End",
        "feat_acoustic_word_duration": "Word_Duration",
        "feat_acoustic_word_gap": "Word_Gap",
    }
    for k, col in {**struct_map, **acoustic_map}.items():
        if not features.get(k, True):
            drop.add(col)
    return drop
```

```python
class Card(QFrame):
    """A white-background, bordered container for grouping a feature subsection."""

    def __init__(self, title: str, subtitle: str = "", parent=None):
```

```python
        return self.api_key_line.text() if self.api_key_row.isVisible() else ""

    def _make_toggle(self, key, label, initial, section):
        row = ToggleRow(label=label)
        row.set_checked(initial)
        row.toggled.connect(lambda v, k=key, s=section: self._on_toggle(s, k, v))
        self._toggles[key] = row
        return row

    def _on_toggle(self, section, key, value):
        prefix = "output_"
        if key.startswith(prefix):
            save_preferences({"output": {key[len(prefix):]: value}})
        else:
            save_preferences({section: {key: value}})
        self.settings_changed.emit()

    def refresh_warnings(self, session_api_key: str = "") -> None:
        from gui.preflight import compute_warnings
        warnings = compute_warnings(session_api_key=session_api_key)
        for row in self._toggles.values():
            row.set_warning(None)
        for w in warnings:
            row = self._toggles.get(w.toggle_key)
            if row is not None:
                row.set_warning(w.message)

    def _on_model_change(self, label):
        model_id = next((mid for mid, lab in LLM_OPTIONS if lab == label), label)
        save_preferences({"predictability": {"llm_model": model_id}})
        self._refresh_api_key_visibility()
        self.settings_changed.emit()

    def _refresh_api_key_visibility(self):
        label = self.llm_model_combo.currentText()
        model_id = next((mid for mid, lab in LLM_OPTIONS if lab == label), label)
        self.api_key_row.setVisible(model_id.startswith("gpt-"))

    def _on_backend_change(self, backend):
        save_preferences({"semantic": {"backend": backend}})
        self._update_cats_count_text()
        self.settings_changed.emit()

    def _cats_count_text(self) -> str:
        prefs = load_preferences()
        backend = prefs["semantic"].get("backend", "auto")
        if backend == "auto":
            from extractors.semantic_domain import detect_backend
            backend, _ = detect_backend()
        key = f"word_categories_{backend}"
        sel = prefs["semantic"].get(key, [])
        return f"{len(sel)} categories selected for word level."

    def _update_cats_count_text(self):
        self.cats_count_label.setText(self._cats_count_text())

    def _open_category_dialog(self):
        from gui.category_dialog import CategoryDialog
        prefs = load_preferences()
        backend = prefs["semantic"].get("backend", "auto")
        if backend == "auto":
            from extractors.semantic_domain import detect_backend
            backend, _ = detect_backend()
        dlg = CategoryDialog(backend=backend, level="word", parent=self)
        dlg.exec()
        self._update_cats_count_text()
        self.settings_changed.emit()
```

```python
    def toggle(self, key: str): return self._toggles.get(key)

    def _make_toggle(self, key, label, initial, section):
        row = ToggleRow(label=label)
        row.set_checked(initial)
        row.toggled.connect(lambda v, k=key, s=section: self._on_toggle(s, k, v))
        self._toggles[key] = row
        return row

    def _on_toggle(self, section, key, value):
        prefix = "output_"
        if key.startswith(prefix):
            save_preferences({"output": {key[len(prefix):]: value}})
        else:
            save_preferences({section: {key: value}})
        self.settings_changed.emit()

    def _cats_count_text(self) -> str:
        prefs = load_preferences()
        backend = prefs["semantic"].get("backend", "auto")
        if backend == "auto":
            from extractors.semantic_domain import detect_backend
            backend, _ = detect_backend()
        key = f"doc_categories_{backend}"
        sel = prefs["semantic"].get(key, [])
        return f"{len(sel)} categories selected for document level."

    def _update_cats_count_text(self):
        self.cats_count_label.setText(self._cats_count_text())

    def _open_category_dialog(self):
        from gui.category_dialog import CategoryDialog
        prefs = load_preferences()
        backend = prefs["semantic"].get("backend", "auto")
        if backend == "auto":
            from extractors.semantic_domain import detect_backend
            backend, _ = detect_backend()
        dlg = CategoryDialog(backend=backend, level="document", parent=self)
        dlg.exec()
        self._update_cats_count_text()
        self.settings_changed.emit()
```
