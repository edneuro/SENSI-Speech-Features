```python
"""Lexical diversity measures (TTR, MTLD).

Operates on a list of lowercased content-word tokens.

TTR (Type-Token Ratio):
    Unique tokens / total tokens. Simple but length-biased.

MTLD (Measure of Textual Lexical Diversity, McCarthy & Jarvis 2010):
    Walks tokens; whenever running TTR drops at-or-below `threshold`
    (default 0.72), counts a factor and resets. Final score is the mean
    of forward and reverse passes. Length-corrected.
"""

import math

_DEFAULT_THRESHOLD = 0.72


def compute_ttr(tokens: list[str]) -> float:
    """Return type-token ratio. NaN for empty input. Case-insensitive."""
    if not tokens:
        return float('nan')
    lowered = [t.lower() for t in tokens]
    return len(set(lowered)) / len(lowered)


def compute_mtld(tokens: list[str], threshold: float = _DEFAULT_THRESHOLD) -> float:
    """Return MTLD (forward+reverse mean). NaN for empty or unscoreable input."""
    if not tokens:
        return float('nan')
    lowered = [t.lower() for t in tokens]
    fwd = _mtld_one_direction(lowered, threshold)
    rev = _mtld_one_direction(list(reversed(lowered)), threshold)
    if math.isnan(fwd) or math.isnan(rev):
        return float('nan')
    return (fwd + rev) / 2.0


def _mtld_one_direction(tokens: list[str], threshold: float) -> float:
    if not tokens:
        return float('nan')
    factor_count = 0
    seen: set[str] = set()
    running_total = 0
    for tok in tokens:
        running_total += 1
        seen.add(tok)
        ttr = len(seen) / running_total
        if ttr <= threshold:
            factor_count += 1
            seen.clear()
            running_total = 0
    # Partial factor for trailing tokens that didn't cross the threshold
    if running_total > 0:
        ttr = len(seen) / running_total
        if ttr < 1.0:
            partial = (1.0 - ttr) / (1.0 - threshold)
        else:
            partial = 0.0
        factor_count += partial
    if factor_count == 0:
        return float('nan')
    return len(tokens) / factor_count
```

```python
"""Readability metrics — Flesch Reading Ease and Flesch-Kincaid Grade Level.

Operates on raw text (preserves the sentence structure that the formulas need).
Uses the `textstat` package, which implements both metrics with the standard
published coefficients.

Empty or whitespace-only text returns NaN for both values, since the formulas
divide by sentence count and word count.
"""

import math


def compute_readability(text: str) -> dict[str, float]:
    """Return {'flesch_reading_ease': float, 'flesch_kincaid_grade': float}."""
    if not text or not text.strip():
        return {
            'flesch_reading_ease': math.nan,
            'flesch_kincaid_grade': math.nan,
        }

    import textstat  # lazy import — heavy dependency
    try:
        ease = float(textstat.flesch_reading_ease(text))
        grade = float(textstat.flesch_kincaid_grade(text))
    except Exception:
        ease = math.nan
        grade = math.nan
    return {
        'flesch_reading_ease': ease,
        'flesch_kincaid_grade': grade,
    }
```

Create `extractors/liwc_summary.py`:
```python
"""LIWC summary scores: Authentic, Analytic, Tone, Clout.

Implements approximations of LIWC-22's four summary indices using the
publicly documented Pennebaker et al. formulas. Results are *close to* but
not guaranteed to match LIWC-22's official numbers exactly; LIWC's
proprietary post-processing (z-score percentile transformation) may
differ. Users who need exact LIWC-22 numbers should run LIWC's desktop
software on their text and compare; for relative comparisons across
stimuli, this implementation is generally adequate.

Citation: Pennebaker, J. W., Chung, C. K., Frazee, J., Lavergne, G. M., &
Beaver, D. I. (2014). When small words foretell academic success.
PLOS ONE, 9(12), e115844.

LIWC categories required for full computation (must be present in the
.dic file): article, prep, pronoun, ipron, ppron, i, we, you, shehe,
they, auxverb, adverb, conj, negate, posemo, negemo, focuspast,
focusfuture, discrep.
"""

import math
import re
from pathlib import Path


EXPECTED_KEYS = {'authentic', 'analytic', 'tone', 'clout'}

_TOKEN_RE = re.compile(r"[A-Za-z']+")


def compute_liwc_summary(text: str, dictionary_path: Path | str) -> dict[str, float]:
    """Return {'authentic', 'analytic', 'tone', 'clout'} as floats.

    Empty text returns all NaN.
    """
    if not text or not text.strip():
        return {k: math.nan for k in EXPECTED_KEYS}

    parser = _load_parser(Path(dictionary_path))
    counts, n_words = _category_counts_from_text(text, parser)
    if n_words == 0:
        return {k: math.nan for k in EXPECTED_KEYS}

    def pct(cat: str) -> float:
        return 100.0 * counts.get(cat, 0) / n_words

    # Analytic — Pennebaker et al. 2014, Table 2
    analytic = (
        30.0
        + pct('article') * 0.8
        + pct('prep') * 0.7
        - pct('pronoun') * 0.4
        - pct('ipron') * 0.4
        - pct('auxverb') * 0.4
        - pct('conj') * 0.4
        - pct('adverb') * 0.4
        - pct('negate') * 0.4
    )

    # Clout — Pennebaker et al. 2014 (positive contributors are second/third
    # person; negative are first-person singular and tentative markers)
    clout = (
        50.0
        + pct('we') * 0.8
        + pct('you') * 0.8
        + pct('shehe') * 0.5
        + pct('they') * 0.5
        - pct('i') * 0.8
        - pct('ipron') * 0.4
    )

    # Authentic — first-person singular + present-focus + cognitive
    # mechanisms increase Authentic; complex hedging decreases it.
    authentic = (
        50.0
        + pct('i') * 0.8
        + pct('discrep') * 0.4
        + pct('focuspresent') * 0.2
        - pct('we') * 0.4
        - pct('focusfuture') * 0.2
    )

    # Tone — LIWC-22 manual: positive emotion minus negative emotion,
    # centered at 50 (neutral).
    tone = 50.0 + pct('posemo') - pct('negemo')

    # Clamp to 0-100 (LIWC's official outputs are reported in that range)
    return {
        'authentic': max(0.0, min(100.0, authentic)),
        'analytic': max(0.0, min(100.0, analytic)),
        'tone': max(0.0, min(100.0, tone)),
        'clout': max(0.0, min(100.0, clout)),
    }


def _category_counts_from_text(text: str, parser) -> tuple[dict[str, int], int]:
    """Tokenize `text` and count category memberships for each token."""
    tokens = _TOKEN_RE.findall(text)
    counts: dict[str, int] = {}
    for tok in tokens:
        for cat in parser(tok.lower()):
            counts[cat] = counts.get(cat, 0) + 1
    return counts, len(tokens)


def _load_parser(dictionary_path: Path):
    """Return a callable that takes a token and returns a list of category names."""
    try:
        import liwc
    except ImportError as exc:
        raise RuntimeError(f'`liwc` package required for liwc_summary: {exc}')
    parse, _ = liwc.load_token_parser(str(dictionary_path))
    return parse
```

```python
        else:
            # v1 flat schema → migrate
            _migrate_v1_into(merged, user)
            # Rewrite on disk so next load is clean
            _write(path, merged)

    return merged


def save_preferences(updates: dict[str, Any]) -> None:
    """Merge `updates` into existing preferences and write to disk."""
    path = get_preferences_path()
    path.parent.mkdir(parents=True, exist_ok=True)
    current = load_preferences()
    _deep_merge_sections(current, updates)
    current['schema_version'] = SCHEMA_VERSION
    _write(path, current)


def _write(path: Path, data: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('w', encoding='utf-8') as f:
        json.dump(data, f, indent=2)


def _deep_merge_sections(into: dict, src: dict) -> None:
    """Merge `src` into `into`. For known top-level dict keys (shared, word_level,
    document_level), merge their inner contents (so unspecified keys keep defaults).
    For other keys, replace wholesale. Preserve any unknown keys from `src`.
    """
    for key, value in src.items():
        if key in ('shared', 'word_level', 'document_level') and isinstance(value, dict):
            existing = into.get(key, {})
            if not isinstance(existing, dict):
                existing = {}
            for sub_key, sub_value in value.items():
                if sub_key == 'feature_toggles' and isinstance(sub_value, dict):
                    base = dict(existing.get('feature_toggles', {}))
                    base.update(sub_value)
                    existing['feature_toggles'] = base
                else:
                    existing[sub_key] = sub_value
            into[key] = existing
        else:
            into[key] = value


def _migrate_v1_into(merged: dict, v1: dict) -> None:
    """Move v1 flat fields into the v2 nested layout in `merged`."""
    shared = merged['shared']
    word = merged['word_level']

    for key in ('analyzer_name', 'last_transcripts_folder',
                'last_audio_folder', 'last_output_folder',
                'dictionary_backend'):
        if key in v1:
            shared[key] = v1[key]

    if 'semantic_categories' in v1:
        word['semantic_categories'] = list(v1['semantic_categories'])
        merged['document_level']['semantic_categories'] = list(v1['semantic_categories'])
    if 'llm_model' in v1:
        word['llm_model'] = v1['llm_model']
    if isinstance(v1.get('feature_toggles'), dict):
        base = dict(_WORD_TOGGLES_DEFAULT)
        for k, v in v1['feature_toggles'].items():
            if k in base:
                base[k] = v
        word['feature_toggles'] = base
```

```python
"""Per-stimulus aggregation — one-row summary for each processed pair.

Reads the per-word DataFrame produced by the extractor pipeline and emits
a flat dict of doc-level fields. Each feature group is controlled by a
toggle key (see DOC_FEATURE_KEYS); toggled-off groups are absent from
the output.

When 'category_proportions' is on, scans Sem_* columns and emits
`count_<cat>` / `prop_<cat>` over content tokens.

When `liwc_*` toggles are requested and `dictionary_used == 'LIWC'`,
adds the corresponding LIWC summary scores using `extractors.liwc_summary`.
"""

from pathlib import Path

import pandas as pd

from extractors.lexical_diversity import compute_ttr, compute_mtld
from extractors.readability import compute_readability

CONTENT_POS = {'NOUN', 'VERB', 'ADJ', 'ADV', 'PROPN'}

DOC_FEATURE_KEYS = (
    'counts',
    'mean_zipf_freq',
    'mean_edu_zipf',
    'mean_surprisal',
    'mean_llm_cloze',
    'ttr',
    'mtld',
    'flesch_reading_ease',
    'flesch_kincaid_grade',
    'category_proportions',
    'liwc_authentic',
    'liwc_analytic',
    'liwc_tone',
    'liwc_clout',
)


def build_doc_summary(
    token_df: pd.DataFrame,
    stimulus_id: str,
    dictionary_used: str,
    text: str,
    toggles: dict[str, bool],
    liwc_dictionary_path: Path | str | None = None,
) -> dict:
    """Aggregate a per-word DataFrame into a one-row summary dict."""
    df = token_df
    summary: dict = {
        'stimulus_id': stimulus_id,
        'dictionary_used': dictionary_used,
    }

    if toggles.get('counts'):
        summary['n_tokens'] = int(len(df))
        summary['n_content_tokens'] = int(df['upos'].isin(CONTENT_POS).sum()) if 'upos' in df else 0
        summary['n_sentences'] = int(df['sentence_id'].nunique()) if 'sentence_id' in df else 0
        summary['n_paragraphs'] = int(df['paragraph_id'].nunique()) if 'paragraph_id' in df else 0

    if toggles.get('mean_zipf_freq') and 'Zipf_Freq' in df.columns:
        summary['mean_zipf_freq'] = _mean(df['Zipf_Freq'])

    if toggles.get('mean_edu_zipf'):
        for src, dst in [
            ('Edu_Zipf_K2', 'mean_edu_zipf_k2'),
            ('Edu_Zipf_3to5', 'mean_edu_zipf_3to5'),
            ('Edu_Zipf_6to8', 'mean_edu_zipf_6to8'),
            ('Edu_Zipf_9to12', 'mean_edu_zipf_9to12'),
            ('Edu_Zipf_Overall', 'mean_edu_zipf_overall'),
        ]:
            if src in df.columns:
                summary[dst] = _mean(df[src])

    if toggles.get('mean_surprisal') and 'Surprisal' in df.columns:
        summary['mean_surprisal'] = _mean(df['Surprisal'])

    if toggles.get('mean_llm_cloze'):
        for src, dst in [
            ('LLM_Cloze', 'mean_llm_cloze'),
            ('LLM_Surprisal', 'mean_llm_surprisal'),
            ('LLM_TopMatch', 'mean_llm_top_match_prop'),
        ]:
            if src in df.columns:
                summary[dst] = _mean(df[src])

    content_tokens = []
    if 'upos' in df.columns and 'token' in df.columns:
        content_tokens = [
            str(t) for t, p in zip(df['token'], df['upos']) if p in CONTENT_POS
        ]

    if toggles.get('ttr'):
        summary['ttr'] = compute_ttr(content_tokens)
    if toggles.get('mtld'):
        summary['mtld'] = compute_mtld(content_tokens)

    if toggles.get('flesch_reading_ease') or toggles.get('flesch_kincaid_grade'):
        read = compute_readability(text)
        if toggles.get('flesch_reading_ease'):
            summary['flesch_reading_ease'] = read['flesch_reading_ease']
        if toggles.get('flesch_kincaid_grade'):
            summary['flesch_kincaid_grade'] = read['flesch_kincaid_grade']

    if toggles.get('category_proportions'):
        sem_cols = [c for c in df.columns
                    if c.startswith('Sem_') and c != 'Sem_Categories']
        n_content = summary.get('n_content_tokens')
        if n_content is None:
            n_content = int(df['upos'].isin(CONTENT_POS).sum()) if 'upos' in df else 0
        for col in sem_cols:
            parts = col.split('_', 2)
            if len(parts) < 3:
                continue
            category = parts[2]
            count = int(df[col].fillna(0).astype(int).sum())
            summary[f'count_{category}'] = count
            summary[f'prop_{category}'] = (count / n_content) if n_content > 0 else 0.0

    liwc_wanted = any(toggles.get(k) for k in
                      ('liwc_authentic', 'liwc_analytic', 'liwc_tone', 'liwc_clout'))
    if liwc_wanted and dictionary_used == 'LIWC' and liwc_dictionary_path:
        from extractors.liwc_summary import compute_liwc_summary
        scores = compute_liwc_summary(text, liwc_dictionary_path)
        for short_key in ('authentic', 'analytic', 'tone', 'clout'):
            full_key = f'liwc_{short_key}'
            if toggles.get(full_key):
                summary[full_key] = scores[short_key]

    return summary


def _mean(series: pd.Series) -> float | None:
    """Mean that skips NaN; returns None if all values are NaN."""
    if series.notna().any():
        return float(series.mean())
    return None
```

```python
"""Modal dialog for selecting semantic-domain categories.

Parameterized by `level` ('word' or 'document') so each tab edits its own
selection in preferences.json under prefs[<level>]['semantic_categories'].

Close behavior:
    - OK button or X-out: save current selection and close
    - Cancel button: discard changes and close
"""

import tkinter as tk
from tkinter import ttk, messagebox

from utils.preferences import DEFAULT_PREFS, load_preferences, save_preferences
from extractors.semantic_domain import detect_backend


def _all_categories(backend: str, dict_path) -> list[str]:
    if backend == 'empath':
        try:
            from empath import Empath
            return sorted(Empath().cats.keys())
        except Exception:
            return list(DEFAULT_PREFS['word_level']['semantic_categories'])
    if backend == 'liwc' and dict_path is not None:
        try:
            import liwc
            _, names = liwc.load_token_parser(str(dict_path))
            return sorted(names)
        except Exception:
            pass
    return list(DEFAULT_PREFS['word_level']['semantic_categories'])


class CategoriesDialog(tk.Toplevel):
    def __init__(self, master, level: str = 'word'):
        super().__init__(master)
        assert level in ('word', 'document'), level
        self._level = level
        section = f'{level}_level'

        title = ('Configure word-level categories' if level == 'word'
                 else 'Configure document-level categories')
        self.title(title)
        self.geometry('420x520')
        self.transient(master)
        self.grab_set()

        backend, dict_path = detect_backend()
        self._backend = backend
        all_cats = _all_categories(backend, dict_path)

        prefs = load_preferences()
        sop_default = DEFAULT_PREFS[section]['semantic_categories']
        selected = set(prefs[section].get('semantic_categories', sop_default))

        # On-close (X) saves
        self.protocol('WM_DELETE_WINDOW', self._save_and_close)

        search_var = tk.StringVar()
        ttk.Label(self, text=f'Backend: {backend}').pack(anchor='w', padx=10, pady=(10, 4))
        ttk.Label(self, text='Search:').pack(anchor='w', padx=10)
        ttk.Entry(self, textvariable=search_var).pack(fill='x', padx=10, pady=(0, 8))

        canvas = tk.Canvas(self, highlightthickness=0)
        scroll = ttk.Scrollbar(self, orient='vertical', command=canvas.yview)
        canvas.configure(yscrollcommand=scroll.set)
        canvas.pack(side='left', fill='both', expand=True, padx=(10, 0), pady=(0, 8))
        scroll.pack(side='right', fill='y', padx=(0, 10), pady=(0, 8))

        frame = ttk.Frame(canvas)
        canvas.create_window((0, 0), window=frame, anchor='nw')

        self._vars: dict[str, tk.BooleanVar] = {}
        self._checkbuttons: dict[str, ttk.Checkbutton] = {}
        for cat in all_cats:
            var = tk.BooleanVar(value=cat in selected)
            self._vars[cat] = var
            cb = ttk.Checkbutton(frame, text=cat, variable=var)
            cb.pack(anchor='w')
            self._checkbuttons[cat] = cb

        frame.update_idletasks()
        canvas.configure(scrollregion=canvas.bbox('all'))

        def _filter(*_):
            q = search_var.get().lower().strip()
            for cat, cb in self._checkbuttons.items():
                if not q or q in cat.lower():
                    cb.pack(anchor='w')
                else:
                    cb.pack_forget()

        search_var.trace_add('write', _filter)

        btn_frame = ttk.Frame(self)
        btn_frame.pack(fill='x', side='bottom', padx=10, pady=10)
        ttk.Button(btn_frame, text='Reset to SOP default',
                   command=self._reset).pack(side='left')
        ttk.Button(btn_frame, text='Cancel',
                   command=self._discard_and_close).pack(side='right')
        ttk.Button(btn_frame, text='OK',
                   command=self._save_and_close).pack(side='right', padx=(0, 6))

    def _reset(self):
        section = f'{self._level}_level'
        defaults = set(DEFAULT_PREFS[section]['semantic_categories'])
        for cat, var in self._vars.items():
            var.set(cat in defaults)

    def _save_and_close(self):
        chosen = [cat for cat, var in self._vars.items() if var.get()]
        if not chosen:
            messagebox.showwarning(
                'No categories selected',
                'Please select at least one category before saving.'
            )
            return
        section = f'{self._level}_level'
        save_preferences({section: {'semantic_categories': chosen}})
        self.destroy()

    def _discard_and_close(self):
        self.destroy()
```

```python
class _LogRedirect:
    def __init__(self, app):
        self._app = app

    def write(self, msg):
        self._app.after(0, self._app.append_log, msg)

    def flush(self):
        pass


def _kick_off(app, state, args, *, level: str) -> None:
    state['run_btn'].configure(state='disabled')
    app.cancel_btn.configure(state='normal')
    app.status_var.set('Running…')
    app._cancel_requested = False
    threading.Thread(target=_thread_run, args=(app, state, args, level),
                     daemon=True).start()


def _thread_run(app, state, args, level: str) -> None:
    redir = _LogRedirect(app)
    old_out, old_err = sys.stdout, sys.stderr
    sys.stdout = sys.stderr = redir
    try:
        from main import run_word_level_batch, run_doc_level_batch

        def progress(done, total, stimulus_id):
            pct = int(100 * done / total) if total else 0
            app.after(0, lambda: app.progress.configure(value=pct))
            app.after(0, lambda: app.status_var.set(
                f'{level}: {stimulus_id}  ({done}/{total})'))

        runner = run_word_level_batch if level == 'word' else run_doc_level_batch
        runner(args, on_pair_done=progress,
               should_cancel=lambda: app._cancel_requested)
        app.after(0, lambda: app.status_var.set('Done'))
        app.after(0, lambda: messagebox.showinfo(
            'Done', f'{level.capitalize()}-level run finished.\nOutput: {args.output_folder}'))
    except Exception as exc:
        msg = str(exc)
        app.after(0, lambda: app.status_var.set('Error'))
        app.after(0, lambda m=msg: messagebox.showerror('Error', m))
    finally:
        sys.stdout, sys.stderr = old_out, old_err
        app.after(0, lambda: state['run_btn'].configure(state='normal'))
        app.after(0, lambda: app.cancel_btn.configure(state='disabled'))
```
